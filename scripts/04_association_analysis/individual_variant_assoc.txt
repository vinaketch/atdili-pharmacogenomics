# Creating Soft Links for VCF Files
# The 'ln -s' command creates symbolic (soft) links for all .vcf.gz files from the original directory
# to the current working directory. This helps avoid duplicating large files.

for dir in /dataA/dili/2026-04-30-sctask0442981-o/dragen/DIL-*; do
    sample=$(basename $dir)
    ln -s ${dir}/${sample}.hard-filtered.vcf.gz .
    ln -s ${dir}/${sample}.hard-filtered.vcf.gz.tbi .
done


#Counting the merge vcf files
ls *.hard-filtered.vcf.gz | wc -l


# Merging VCF files
# 'bcftools merge' combines multiple VCF files (file1, file2, ... file96) into a single merged VCF file.
# -m none: disables multi-allelic site merging
# -O z: outputs in compressed VCF format (.vcf.gz)
# -o: specifies output file name
bcftools merge -m none -O z -o merged.vcf.gz *.hard-filtered.vcf.gz

#view merged vcf files
less -S merged.vcf.gz

#Creating sample list from the vcf file
bcftools query -l merged.vcf.gz > sample_list.txt


#Copying the sample list file from server to my local computer, type this on the local computer
scp aketchv@ui.core.wits.ac.za:/home/aketchv/dili/vcfs/sample_list.txt "/mnt/c/School work/PhD project/ATDILI_PGX/Objective 2_Genomics/patients_metadata/"



# Filtering
# Filters variants based on quality and allele depth using bcftools.
# -s LowQual: labels filtered records as LowQual
# '-i QUAL>=30 && AD[*:1]>=15': includes only records with quality >=30 and allele depth >=15
# -g8 -G10: set genotype and sample filtering thresholds
# -Oz: output compressed VCF format
bcftools filter -s LowQual '-i QUAL>=30 && AD[*:1]>=15' -g8 -G10 -Oz -o merged_filtered_pass.vcf.gz merged.vcf.gz

#Index the merge file
tabix merged_filtered_pass.vcf.gz


# Preparing VEP input file
# Removes patient/sample identifiers from the VCF to create a de-identified version for annotation.
# The header and body are processed separately, then combined and compressed.
bcftools view -h merged_filtered_pass.vcf.gz | grep -v '^#CHROM' > header.vcf
bcftools view -H merged_filtered_pass.vcf.gz | cut -f1-9 > body.vcf
cat header.vcf <(echo -e "#CHROM\tPOS\tID\tREF\tALT\tQUAL\tFILTER\tINFO\tFORMAT") body.vcf | bgzip > vep_input_deidentified.vcf.gz
tabix -p vcf vep_input_deidentified.vcf.gz

Calculating AF
bcftools +fill-tags   vep_input_deidentified.vcf.gz     -- -t AC | bcftools query     -f '%CHROM\t%POS\t%REF\t%ALT\t%AC\t572\n' > dili_286_AC.txt

awk 'BEGIN{OFS="\t"} {print $1,$2,$3,$4,$5,$6,$5/$6}' dili_286_AC.txt > dili_286_AF.txt

LOG TRANSFORMED LINEAR ASSOC OF 15 GENES WITH COVARIATES

#!/bin/bash

# ==========================================================
# 15 GENES - LOG TRANSFORMED ASSOCIATION ANALYSIS
# WITH COVARIATES (gender, alcohol, hiv_status)
# ==========================================================

# =========================
# 1. RUN PLINK2 ANALYSIS
# =========================

plink2 \
    --pfile dili_15genes_final \
    --pheno phenotype.txt \
    --pheno-name ln_lb_alt \
    --covar phenotype.txt \
    --covar-name gender,alcohol,hiv_status \
    --glm \
    --adjust \
    --out dili_15genes_ln_lbalt_covar


# =========================
# 2. CHECK OUTPUT COLUMNS
# =========================

head -n 1 dili_15genes_ln_lbalt_covar.ln_lb_alt.glm.linear | tr '\t' '\n' | nl


# =========================
# 3. EXTRACT ADDITIVE MODEL ONLY
# =========================

awk 'NR==1 || $10=="ADD"' \
dili_15genes_ln_lbalt_covar.ln_lb_alt.glm.linear \
> dili_15genes_ln_lbalt_covar_ADD.txt


# =========================
# 4. EXTRACT SNP IDS
# =========================

awk 'NR>1 {print $3}' \
dili_15genes_ln_lbalt_covar_ADD.txt \
> add_ids_15genes.txt


# =========================
# 5. MAP TO ADJUSTED RESULTS
# =========================

awk 'NR==FNR {ids[$1]; next} NR==1 || ($2 in ids)' \
add_ids_15genes.txt \
dili_15genes_ln_lbalt_covar.ln_lb_alt.glm.linear.adjusted \
> dili_15genes_ln_lbalt_covar_ADD_adjusted.txt


# =========================
# 6. NOMINAL SIGNIFICANCE (P < 0.05)
# =========================

awk 'NR==1 || $15 < 0.05' \
dili_15genes_ln_lbalt_covar_ADD.txt \
| column -t | less -S


# =========================
# 7. SORT ALL RESULTS BY P-VALUE
# =========================

{
    head -1 dili_15genes_ln_lbalt_covar_ADD.txt
    tail -n +2 dili_15genes_ln_lbalt_covar_ADD.txt | sort -g -k15,15
} > dili_15genes_ln_lbalt_covar_ADD_sorted_by_P.txt


# =========================
# 8. SORT SIGNIFICANT SNPs ONLY (P < 0.05)
# =========================

{
    head -1 dili_15genes_ln_lbalt_covar_ADD.txt
    awk 'NR>1 && $15<0.05' dili_15genes_ln_lbalt_covar_ADD.txt | sort -g -k15,15
} > dili_15genes_ln_lbalt_covar_ADD_sig_unadjusted_sorted.txt


# =========================
# 9. BONFERRONI SIGNIFICANCE (ADJUSTED FILE)
# =========================

{
    head -1 dili_15genes_ln_lbalt_covar_ADD_adjusted.txt
    awk 'NR>1 && $6 < 0.05' dili_15genes_ln_lbalt_covar_ADD_adjusted.txt | sort -g -k6,6
} > dili_15genes_ln_lbalt_covar_ADD_sig_bonferroni_sorted.txt


# =========================
# DONE
# =========================

echo "15-gene ln-transformed association pipeline completed successfully."
